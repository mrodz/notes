\begin{equation}
\begin{split}
\text{Given:}&\quad (f: A\to B)\text{ is surjective onto B}\\
&\quad (g: B\to C)\text{ is surjective onto C}\\
\\
\text{Prove:}&\quad (g \circ f) \text{ is surjective} \\
\\
&\quad \text{For surjectivity, } \forall c \in C, \exists f(a) \in B : g(f(a))=c \\
\text{Steps:}&\quad \text{By surjectivity, }\forall c \in C, \exists b \in B : g(b)=c \\
\\
&\quad \text{We can show } \forall c \in C,\exists f(a) \in B \text{ if } f(a)\subseteq B\\
&\quad \text{By surjectivity, } \forall b \in B, \exists a \in A : f(a)=b\\
&\quad \text{Hence, } f(a) \subseteq B \\
\\
&\quad \text{Thus, } g(f(a))=c \implies (g\circ f) \text{ is surjective } \blacksquare
\end{split}
\end{equation}
